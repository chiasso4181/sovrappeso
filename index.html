function generaPDFDiretto() { // Prepariamo gli elementi della pagina const element = document.getElementById('contenuto-report'); const btnStampa = document.getElementById('btn-stampa'); const header = document.getElementById('header-stampa'); const footer = document.getElementById('footer-stampa'); const police = document.getElementById('police-stampa'); // Mostriamo temporaneamente gli elementi ufficiali e nascondiamo il pulsante btnStampa.style.display = 'none'; header.style.display = 'flex'; footer.style.display = 'block'; police.style.display = 'flex'; // TRUCCO: Forziamo la larghezza del contenitore come se fosse un PC desktop // Questo impedisce agli elementi di andare a capo e allungare la pagina const larghezzaOriginale = element.style.width; const maxLarghezzaOriginale = element.style.maxWidth; element.style.width = '800px'; element.style.maxWidth = '800px'; // Assicuriamoci che i valori scritti negli input vengano trasferiti correttamente al PDF const inputs = document.querySelectorAll('input, select'); inputs.forEach(input => { input.setAttribute('value', input.value); }); // Opzioni aggiornate per forzare 1 pagina A4 const opt = { margin: 5, // Margini ridotti (5mm) filename: 'Report_Sovraccarico.pdf', image: { type: 'jpeg', quality: 0.98 }, // Aggiunto 'windowWidth: 800' per far credere al sistema di essere su uno schermo grande html2canvas: { scale: 2, useCORS: true, windowWidth: 800 }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' } }; // Creiamo e scarichiamo il PDF html2pdf().set(opt).from(element).save().then(() => { // Dopo aver salvato il PDF, ripristiniamo tutto alla normalità per il telefono btnStampa.style.display = 'block'; header.style.display = 'none'; footer.style.display = 'none'; police.style.display = 'none'; // Ripristiniamo la larghezza originale adatta al telefono element.style.width = larghezzaOriginale; element.style.maxWidth = maxLarghezzaOriginale; }); }
